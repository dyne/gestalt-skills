cmake_minimum_required(VERSION 3.20)
project(modern_imgui_native LANGUAGES C CXX)
enable_testing()
set(IMGUI_DIR "" CACHE PATH "Compatible local Dear ImGui checkout (19297+ API)")
set(GLFW_DIR "" CACHE PATH "Local GLFW checkout; empty uses a configured glfw3 package")
set(IMKIT_DIR "" CACHE PATH "Optional compatible local ImKit checkout")
foreach(file imgui.cpp imgui_draw.cpp imgui_widgets.cpp imgui_tables.cpp backends/imgui_impl_glfw.cpp backends/imgui_impl_opengl3.cpp)
  if(NOT EXISTS "${IMGUI_DIR}/${file}")
    message(FATAL_ERROR "Missing local ImGui source: ${file}; no dependencies are downloaded")
  endif()
endforeach()
find_package(OpenGL REQUIRED)
if(GLFW_DIR)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(GLFW_BUILD_X11 ON CACHE BOOL "" FORCE)
    set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
  endif()
  add_subdirectory("${GLFW_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/glfw" EXCLUDE_FROM_ALL)
else()
  find_package(glfw3 3.3 REQUIRED)
endif()
add_library(native_imgui STATIC "${IMGUI_DIR}/imgui.cpp" "${IMGUI_DIR}/imgui_draw.cpp" "${IMGUI_DIR}/imgui_widgets.cpp" "${IMGUI_DIR}/imgui_tables.cpp")
target_include_directories(native_imgui PUBLIC "${IMGUI_DIR}" "${IMGUI_DIR}/backends")
target_compile_features(native_imgui PUBLIC cxx_std_20)
add_library(native_backends STATIC "${IMGUI_DIR}/backends/imgui_impl_glfw.cpp" "${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp")
target_link_libraries(native_backends PUBLIC native_imgui glfw OpenGL::GL)
add_executable(native_example native_host.cpp main.cpp ../settings_panel.cpp)
target_include_directories(native_example PRIVATE ..)
target_link_libraries(native_example PRIVATE native_backends)
if(IMKIT_DIR)
  set(IMKIT_IMGUI_TARGET native_imgui)
  set(IMKIT_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
  set(IMKIT_BUILD_DESIGN_GALLERY OFF CACHE BOOL "" FORCE)
  set(IMKIT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${IMKIT_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/imkit")
  target_sources(native_example PRIVATE ../headless/imkit_panel.cpp)
  target_include_directories(native_example PRIVATE ../headless)
  target_compile_definitions(native_example PRIVATE NATIVE_WITH_IMKIT)
  target_link_libraries(native_example PRIVATE imkit::imkit)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  set(scripts "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts")
  set(artifacts "${CMAKE_CURRENT_BINARY_DIR}/native-evidence")
  add_test(NAME native.lifecycle COMMAND "${Python3_EXECUTABLE}" -B "${scripts}/run_owned_display.py" --output "${artifacts}/lifecycle" -- "$<TARGET_FILE:native_example>" --smoke "{output}/captures")
  add_test(NAME native.relocated_entry COMMAND "${Python3_EXECUTABLE}" -B "${scripts}/run_owned_display.py" --output "${artifacts}/entry" -- "${Python3_EXECUTABLE}" -B "${scripts}/smoke_native_entry.py" --binary "$<TARGET_FILE:native_example>" --output "{output}/entry-result")
  set_tests_properties(native.lifecycle native.relocated_entry PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
endif()
